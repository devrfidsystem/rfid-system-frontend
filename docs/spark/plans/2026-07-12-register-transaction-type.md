# Register as a Transaction Document Type Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use spark:subagent-driven-development (recommended) or spark:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Re-architect "Register" (tag registration) from a bespoke RFID/EPC feature into a proper Transaction document type — a new `transactionKey: "register"` reusing the existing generic List/Create/Detail infrastructure, with 4 header fields only (Document Number, Date Issue, User, Notes) and no line items.

**Architecture:** Backend gets a new `RegisterDoc` Prisma model (header-only, mirrors `OpnameDoc`'s shape) and a new `src/modules/warehouse/register/` module with a simple draft→posted/canceled lifecycle (mirrors `InboundModule`'s `post`/`cancel`, minus any stock-balance mutation). Frontend adds `"register"` everywhere `TransactionKey` branches (services, router, report config), adds an `isRegister` branch to the shared Create/Detail composables and pages (mirrors the existing `isOpname` branch), and retires the old RFID-specific pages. Both repos are independent git repositories — commit separately in each.

**Tech Stack:** NestJS + Prisma + PostgreSQL (Warehouse-be); Vue 3 + TypeScript + vee-validate (Warehouse).

## Global Constraints

- Backend repo: `/Users/syillaeltaniadaffa/Documents/Warehouse-be`. Frontend repo: `/Users/syillaeltaniadaffa/Documents/Warehouse`. These are separate git repositories — commit each repo's changes with its own `git commit`, never mix files from both repos in one commit.
- `RegisterDoc` has exactly 4 user-facing fields: `docNumber` (required, unique per company), `docDate` (required), `registeredById` (required, FK to `users` — the staff who performed the physical registration), `notes` (optional). No line items, no warehouse field, no partner field.
- Status lifecycle: `draft` → `posted` (via `POST /register/:id/post`) or `draft` → `canceled` (via `POST /register/:id/cancel`, draft-only). Posting is a **pure status flip with no stock-balance mutation** — do not copy Inbound's `stockService.applyInboundMovement` call.
- Status string values are lowercase: `draft`, `posted`, `canceled` (American spelling, one `l`) — matches `src/shared/constants/doc-status.constants.ts`'s `DocStatus` type exactly.
- Do not modify `rfid.service.ts`, `rfid.api.ts`, `EpcTag`/`EpcEvent` Prisma models, or any `/rfid/*` backend endpoint — they stay as dead code, unused by any frontend page after this plan, but not deleted.
- Do not add Export-to-XLS support for `register` in this plan — the Export button must be hidden for this transaction type, not wired to a non-existent backend case.
- No data relationship between a `RegisterDoc` and any `InboundDoc` — "Register → Inbound" is process ordering in the sidebar only.

- [ ] **Step 1: Add the `RegisterDoc` model to `prisma/schema.prisma`**

Add this model block. A good place is directly after the `OpnameEpcLine` model (end of the Opname block), so all document-type models stay grouped:

```prisma
model RegisterDoc {
  id             String    @id @db.Uuid
  companyId      String    @map("company_id") @db.Uuid
  docNumber      String    @map("doc_number") @db.VarChar
  docDate        DateTime  @map("doc_date") @db.Timestamptz(6)
  registeredById String    @map("registered_by_id") @db.Uuid
  notes          String?
  status         String    @default("draft") @db.VarChar
  createdById    String?   @map("created_by") @db.Uuid
  createdAt      DateTime? @map("created_at") @db.Timestamptz(6)
  updatedAt      DateTime? @updatedAt @map("updated_at") @db.Timestamptz(6)

  company      Company @relation(fields: [companyId], references: [id], onDelete: NoAction, onUpdate: NoAction)
  registeredBy User    @relation("RegisterDocRegisteredBy", fields: [registeredById], references: [id], onDelete: NoAction, onUpdate: NoAction)
  createdBy    User?   @relation("RegisterDocCreatedBy", fields: [createdById], references: [id], onDelete: NoAction, onUpdate: NoAction)

  @@unique([companyId, docNumber], map: "register_docs_company_id_doc_number_idx")
  @@index([companyId])
  @@index([status])
  @@index([createdAt])
  @@index([registeredById])
  @@map("register_docs")
}
```

Note `docDate` uses `@db.Timestamptz(6)` (matching `InboundDoc.inbound_date`'s type), not `@db.Date` — this keeps it consistent with how every other doc type stores its date, and lets the existing `IsDateString()` DTO validator work unchanged.

```typescript
import { Injectable } from "@nestjs/common";
import { randomUUID } from "node:crypto";
import { PrismaService } from "../../../shared/prisma/prisma.service";
import {
    BusinessRuleException,
    ConflictException,
    NotFoundException,
} from "../../../shared/exceptions/app.exception";
import { buildMeta } from "../../../shared/utils/response.util";
import { buildDateRangeWhere } from "../../../shared/utils/query.util";
import {
    canPostDoc,
    isValidDocStatus,
} from "../../../shared/constants/doc-status.constants";
import type { DocStatus } from "../../../shared/constants/doc-status.constants";
import type { CreateRegisterDocDto } from "./dto/register-doc.dto";
import type { RegisterListFilterDto } from "./dto/register-list-filter.dto";
import type { PaginatedResult } from "../../../shared/interfaces/api-response.interface";
import type { RequestUser } from "../../external/supabase/supabase.types";
import type { Prisma } from "@prisma/client";

const REGISTER_DOC_INCLUDE = {
    company: { select: { id: true, code: true, name: true } },
    registeredBy: { select: { id: true, fullName: true } },
} as const;

@Injectable()
export class RegisterService {
    constructor(private readonly prisma: PrismaService) {}

    async create(dto: CreateRegisterDocDto, user: RequestUser) {
        const company = await this.prisma.company.findUnique({
            where: { id: dto.companyId },
        });
        if (!company) throw new NotFoundException("Company", dto.companyId);

        const registeredBy = await this.prisma.user.findUnique({
            where: { id: dto.registeredById },
        });
        if (!registeredBy)
            throw new NotFoundException("User", dto.registeredById);

        const existing = await this.prisma.registerDoc.findUnique({
            where: {
                companyId_docNumber: {
                    companyId: dto.companyId,
                    docNumber: dto.docNumber,
                },
            },
        });
        if (existing) {
            throw new ConflictException(
                `Register document with number '${dto.docNumber}' already exists for this company`,
            );
        }

        return this.prisma.registerDoc.create({
            data: {
                id: randomUUID(),
                companyId: dto.companyId,
                docNumber: dto.docNumber,
                docDate: new Date(dto.docDate),
                registeredById: dto.registeredById,
                notes: dto.notes ?? null,
                status: "draft",
                createdById: user.id,
            },
            include: REGISTER_DOC_INCLUDE,
        });
    }

    async getDetail(id: string) {
        const doc = await this.prisma.registerDoc.findUnique({
            where: { id },
            include: REGISTER_DOC_INCLUDE,
        });
        if (!doc) throw new NotFoundException("Register document", id);
        return doc;
    }

    async list(
        query: RegisterListFilterDto,
    ): Promise<PaginatedResult<unknown>> {
        const { skip, take } = this.prisma.getPaginationArgs(
            query.page,
            query.limit,
        );

        const where: Prisma.RegisterDocWhereInput = {};
        if (query.companyId) where.companyId = query.companyId;
        if (query.status && isValidDocStatus(query.status))
            where.status = query.status;
        if (query.docNumber)
            where.docNumber = {
                contains: query.docNumber,
                mode: "insensitive",
            };
        if (query.registeredById) where.registeredById = query.registeredById;
        if (query.dateFrom || query.dateTo) {
            where.createdAt = buildDateRangeWhere(query.dateFrom, query.dateTo);
        }

        const [items, total] = await this.prisma.$transaction([
            this.prisma.registerDoc.findMany({
                where,
                skip,
                take,
                orderBy: { createdAt: "desc" },
                include: REGISTER_DOC_INCLUDE,
            }),
            this.prisma.registerDoc.count({ where }),
        ]);

        return { items, meta: buildMeta(total, query.page, query.limit) };
    }

    async post(id: string) {
        const doc = await this.prisma.registerDoc.findUnique({ where: { id } });
        if (!doc) throw new NotFoundException("Register document", id);
        if (!canPostDoc(doc.status as DocStatus)) {
            throw new BusinessRuleException(
                `Document is not in draft status. Current status: ${doc.status}. Only draft can be posted.`,
            );
        }

        return this.prisma.registerDoc.update({
            where: { id },
            data: { status: "posted" },
            include: REGISTER_DOC_INCLUDE,
        });
    }

    async cancel(id: string) {
        const doc = await this.prisma.registerDoc.findUnique({ where: { id } });
        if (!doc) throw new NotFoundException("Register document", id);
        if (doc.status !== "draft") {
            throw new BusinessRuleException(
                "Only draft register documents can be canceled. Posted documents require a reversal flow.",
            );
        }

        return this.prisma.registerDoc.update({
            where: { id },
            data: { status: "canceled" },
            include: REGISTER_DOC_INCLUDE,
        });
    }
}
```

import {
    Body,
    Controller,
    Get,
    Param,
    ParseUUIDPipe,
    Post,
    Query,
} from "@nestjs/common";
import {
    ApiTags,
    ApiOperation,
    ApiParam,
    ApiBody,
    ApiBadRequestResponse,
    ApiNotFoundResponse,
    ApiUnauthorizedResponse,
    ApiForbiddenResponse,
} from "@nestjs/swagger";
import { RegisterService } from "./register.service";
import { CreateRegisterDocDto } from "./dto/register-doc.dto";
import { RegisterListFilterDto } from "./dto/register-list-filter.dto";
import { CurrentUser } from "../../auth/decorators/current-user.decorator";
import {
    paginatedResponse,
    successResponse,
} from "../../../shared/utils/response.util";
import {
    ApiBearerAuthProtected,
    ApiPaginatedOkResponse,
    ApiStandardOkResponse,
} from "../../../shared/swagger/index";
import type { ApiResponse } from "../../../shared/interfaces/api-response.interface";
import type { RequestUser } from "../../external/supabase/supabase.types";

@ApiTags("Register")
@Controller("register")
export class RegisterController {
    constructor(private readonly registerService: RegisterService) {}

    @Post()
    @ApiBearerAuthProtected()
    @ApiOperation({ summary: "Create draft register document" })
    @ApiBody({ type: CreateRegisterDocDto })
    @ApiStandardOkResponse("Register document created")
    @ApiBadRequestResponse()
    @ApiUnauthorizedResponse()
    @ApiForbiddenResponse()
    async create(
        @Body() dto: CreateRegisterDocDto,
        @CurrentUser() user: RequestUser,
    ): Promise<ApiResponse<unknown>> {
        const doc = await this.registerService.create(dto, user);
        return successResponse(doc, "Register document created");
    }

    @Get()
    @ApiBearerAuthProtected()
    @ApiOperation({
        summary: "List register documents (paginated)",
        description:
            "Filter by date range, companyId, status (draft/posted/canceled), document number, and registeredById.",
    })
    @ApiPaginatedOkResponse("Paginated list of register documents")
    @ApiUnauthorizedResponse()
    @ApiForbiddenResponse()
    async list(
        @Query() query: RegisterListFilterDto,
    ): Promise<ApiResponse<unknown>> {
        const result = await this.registerService.list(query);
        return paginatedResponse(result);
    }

    @Get(":id")
    @ApiBearerAuthProtected()
    @ApiOperation({ summary: "Get register document detail" })
    @ApiStandardOkResponse("Register document")
    @ApiParam({
        name: "id",
        format: "uuid",
        description: "Register document ID",
    })
    @ApiNotFoundResponse()
    @ApiUnauthorizedResponse()
    @ApiForbiddenResponse()
    async getDetail(
        @Param("id", ParseUUIDPipe) id: string,
    ): Promise<ApiResponse<unknown>> {
        const doc = await this.registerService.getDetail(id);
        return successResponse(doc);
    }

    @Post(":id/post")
    @ApiBearerAuthProtected()
    @ApiOperation({
        summary: "Post register document",
        description:
            "Status transition: draft → posted. Pure status flip — no stock-balance mutation. Document must be in draft status.",
    })
    @ApiParam({
        name: "id",
        format: "uuid",
        description: "Register document ID",
    })
    @ApiStandardOkResponse("Register document posted")
    @ApiBadRequestResponse()
    @ApiNotFoundResponse()
    @ApiUnauthorizedResponse()
    @ApiForbiddenResponse()
    async post(
        @Param("id", ParseUUIDPipe) id: string,
    ): Promise<ApiResponse<unknown>> {
        const doc = await this.registerService.post(id);
        return successResponse(doc, "Register document posted");
    }

    @Post(":id/cancel")
    @ApiBearerAuthProtected()
    @ApiOperation({ summary: "Cancel register document" })
    @ApiParam({
        name: "id",
        format: "uuid",
        description: "Register document ID",
    })
    @ApiStandardOkResponse("Register document canceled")
    @ApiBadRequestResponse()
    @ApiNotFoundResponse()
    @ApiUnauthorizedResponse()
    @ApiForbiddenResponse()
    async cancel(
        @Param("id", ParseUUIDPipe) id: string,
    ): Promise<ApiResponse<unknown>> {
        const doc = await this.registerService.cancel(id);
        return successResponse(doc, "Register document canceled");
    }
}
